## 先读：这次究竟会简化什么

会减少 DTO 包装类、第三方适配、Swagger 清理以及部分控制器里的手动 `parse`。字段约束、权限控制、Service 业务逻辑和错误脱敏仍然需要保留。

三种原生能力分别长这样，后文会把它们接进你的真实文件：

```ts
// 请求：参数直接绑定 Zod Schema，不再通过 createZodDto 包装类。
login(@Body({ schema: loginSchema }) body: LoginRequest) { /* 业务调用 */ }

// 响应：为这个方法声明返回值校验和字段筛选规则。
@SerializeOptions({ schema: currentUserSchema })

// Swagger：同一份响应 Schema 用于生成文档。
@ApiOkResponse({ standardSchema: currentUserSchema })
```

注意最后一行使用 `standardSchema`。Swagger 的 `schema` 属性接受 OpenAPI Schema 对象，不能直接塞入 Zod 对象。

本文完成两层工作：先完整清除 `nestjs-zod`，接好 Nest 12 原生基础能力；再将认证五个接口的请求、响应与 Swagger 配齐，并把项目控制器里的五处手动校验换成参数 Schema。其余模块现有手动校验仍可运行，通过同一套异常转换返回错误。其他业务的全部响应 Schema 需要按各自公开字段契约继续推广，不会因为安装 Nest 12 自动生效。

**执行规则：按第 1～18 步顺序进行。第 3～12 步是一组连续修改，中途不要启动、构建或提交为可部署版本。编辑器暂时报错是因为依赖仍为 11；完成第 12 步安装后，第 13 步统一检查。每步的检查失败，就停在那一步处理。**

## 第 3 步：准备统一的校验错误详情转换

原生 Pipe 默认抛 `BadRequestException`。你现有 Filter 会把它转成 `COMMON.BAD_REQUEST`，字段详情也会丢失。因此要用自定义 `exceptionFactory` 返回原有 `AppException`。

```ts
/** 校验器问题的最小公共结构，兼容 Standard Schema 与 Zod issues。 */
export interface SchemaValidationIssue {
  readonly message: string
  readonly path?: readonly (PropertyKey | { readonly key: PropertyKey })[]
  readonly code?: string
}
```

打开 `packages/shared/src/types/index.ts`，将原来的 common 导出改为：

```ts

```

### 3.2 新建公共转换函数

新建 `packages/shared/src/utils/schema-validation.ts`，完整内容：

```ts
import type { ApiErrorDetail, SchemaValidationIssue } from '../types/common'

/**
 * 把校验问题映射为统一错误详情，兼容普通路径和 Standard Schema 的 key 路径。
 * 保留校验器提供的错误码；缺少错误码时回退为 invalid。
 * 数量限制、控制字符清理和公开文本长度限制由 Server 全局异常过滤器负责。
 */
export function toValidationDetails(issues: readonly SchemaValidationIssue[]): ApiErrorDetail[] {
  return issues.map((issue) => {
    const path = issue.path
      ?.map((segment) => String(typeof segment === 'object' ? segment.key : segment))
      .join('.')

    return {
      ...(path ? { path } : {}),
      code: typeof issue.code === 'string' ? issue.code : 'invalid',
      message: issue.message,
    }
  })
}
```

打开 `packages/shared/src/utils/index.ts`，末尾追加：

```ts
export { toValidationDetails } from './schema-validation'
```

**检查：**common 的类型导出和 utils 的函数导出各只有一份。这个函数不依赖 Nest，不负责返回 HTTP 响应；Zod 的 `too_small`、`custom` 等 code 继续保留。

```ts
/**
 * 校验并转换请求数据，保留默认值、规范化与严格对象语义。
 * @throws 校验失败时抛出统一参数错误，由全局 Filter 限制和清理公开详情。
 */
export function parse<T>(schema: z.ZodType<T>, input: unknown): T {
  const result = schema.safeParse(input)
  if (!result.success) {
    throw new AppException(COMMON_ERRORS.VALIDATION_FAILED, {
      details: toValidationDetails(result.error.issues),
    })
  }
  return result.data
}
```

```ts
  ChangePasswordResult,

  LoginResult,
  LogoutResult,
  RegisterResult,

/** 用户公开字段白名单；禁止在这里加入 passwordHash 等内部字段。 */
export const authUserSchema = z.object({

  account: z.string(),
  name: z.string(),

  terminal: z.enum(['web', 'desktop', 'mobile']),

export const logoutResultSchema = z.object({
```

## 第 6 步：修改认证控制器和 Service 类型

### 6.1 完整替换 auth.controller.ts

文件：`apps/server/src/modules/auth/auth.controller.ts`。

```ts
import { CurrentAuth } from '@/common/decorators/current-auth.decorator'
import { Authenticated } from '@/common/decorators/access-policy.decorator'
import { Public } from '@/common/decorators/public.decorator'
import {
  Body,
  Controller,
  Get,
  Headers,
  HttpCode,
  HttpStatus,
  Post,
  Req,
  SerializeOptions,
} from '@nestjs/common'
import {
  ApiBearerAuth,
  ApiCreatedResponse,
  ApiOkResponse,
  ApiOperation,
  ApiTags,
} from '@nestjs/swagger'
import type { FastifyRequest } from 'fastify'
import type {
  ChangePasswordRequest,
  ChangePasswordResult,
  CurrentUser,
  LoginRequest,
  LoginResult,
  LogoutResult,
  RegisterRequest,
  RegisterResult,
} from 'shared/types'
import { AuthService } from './auth.service'
import { loginSchema } from './dto/login.dto'
import { registerSchema } from './dto/register.dto'
import { changePasswordSchema } from './password/change-password.dto'
import {
  changePasswordResultSchema,
  currentUserSchema,
  loginResultSchema,
  logoutResultSchema,
  registerResultSchema,
} from './dto/auth-response.dto'
import type { CurrentAuthType } from './session/session.types'

@ApiTags('认证')
@Controller('auth')
export class AuthController {
  constructor(private readonly authService: AuthService) {}

  /** 接收注册资料，返回新账号的公开信息。 */
  @Public()
  @ApiOperation({ summary: '注册' })
  @ApiCreatedResponse({ standardSchema: registerResultSchema })
  @SerializeOptions({ schema: registerResultSchema })
  @Post('register')
  register(@Body({ schema: registerSchema }) body: RegisterRequest): Promise<RegisterResult> {
    return this.authService.register(body)
  }

  /** 从请求提取登录来源，创建或替换当前终端会话。 */
  @Public()
  @ApiOperation({ summary: '登录并创建 Redis Session' })
  @ApiOkResponse({ standardSchema: loginResultSchema })
  @SerializeOptions({ schema: loginResultSchema })
  @HttpCode(HttpStatus.OK)
  @Post('login')
  login(
    @Body({ schema: loginSchema }) body: LoginRequest,
    @Req() request: FastifyRequest,
  ): Promise<LoginResult> {
    return this.authService.login(body, {
      ip: request.ip,
      userAgent: request.headers['user-agent'] ?? '',
    })
  }

  /** 撤销请求头令牌对应的当前终端会话；保持退出接口幂等。 */
  @ApiBearerAuth('session')
  @ApiOperation({ summary: '退出当前端' })
  @ApiOkResponse({ standardSchema: logoutResultSchema })
  @SerializeOptions({ schema: logoutResultSchema })
  @Public()
  @HttpCode(HttpStatus.OK)
  @Post('logout')
  logout(@Headers('authorization') authorization: string | undefined): Promise<LogoutResult> {
    return this.authService.logout(authorization)
  }

  /** 结合认证上下文与最新公开资料，返回当前用户及登录终端。 */
  @ApiBearerAuth('session')
  @ApiOperation({ summary: '使用 Session Token 获取当前用户资料' })
  @ApiOkResponse({ standardSchema: currentUserSchema })
  @SerializeOptions({ schema: currentUserSchema })
  @Get('me')
  @Authenticated()
  async me(@CurrentAuth() auth: CurrentAuthType): Promise<CurrentUser> {
    const user = await this.authService.getCurrentUser(auth.userId)
    return { ...user, terminal: auth.terminal }
  }

  /** 修改本人密码，成功后撤销该用户全部终端会话。 */
  @ApiBearerAuth('session')
  @ApiOperation({ summary: '修改本人密码并撤销全部终端会话' })
  @ApiOkResponse({ standardSchema: changePasswordResultSchema })
  @SerializeOptions({ schema: changePasswordResultSchema })
  @Post('change-password')
  @HttpCode(HttpStatus.OK)
  @Authenticated()
  changePassword(
    @CurrentAuth() auth: CurrentAuthType,
    @Body({ schema: changePasswordSchema }) body: ChangePasswordRequest,
  ): Promise<ChangePasswordResult> {
    return this.authService.changePassword(auth.userId, body)
  }
}
```

`@SerializeOptions` 负责运行时响应处理，`@ApiOkResponse` 负责文档；两者都需要。只加 Swagger 装饰器不会过滤字段，只装全局拦截器也不会自动推断每个接口的 Schema。

### 15.2 在 Swagger 连续做一遍认证

只在自己的本地测试环境使用明显虚构的测试账号，不用真实个人资料。每一步通过再做下一步：

1. POST `/auth/register`：name 用 `示例用户`，account 用未使用过的 `schemaqa_1013`，password 用示例值 `qaOld12345`。预期 201，只有 id／account／name；如果账号已存在，换一个符合规则的新账号。
2. POST `/auth/login`：用刚才的账号密码。预期 200，四个登录结果字段符合第 14 步列表；成功正文没有 `{ code, data, message }` 外壳。
3. 点击 Swagger 的 Authorize，将刚拿到的 accessToken 填到 session Bearer 输入框，通常无需自行加 `Bearer ` 前缀。
4. GET `/auth/me`：预期 200，仅公开用户字段和 terminal；不存在 passwordHash、数据库状态或额外内部字段。
5. POST `/auth/change-password`：oldPassword 用 `qaOld12345`，newPassword 用 `qaNew12345`，confirmPassword 故意用 `different`。预期 400，details.path 含 confirmPassword，code 为 custom；再次请求 me 仍应成功。
6. 重新提交改密，把 confirmPassword 改为 `qaNew12345`。预期 200、`{ "passwordChanged": true }`。旧 token 请求 me 应返回 401，因为你原来的业务逻辑会撤销全部终端会话。
7. 用新密码登录，更新 Authorize，me 恢复 200；用旧密码登录失败。
8. POST `/auth/logout`：预期 200、`{ "loggedOut": true }`。退出后 me 为 401；再次退出仍是成功。

**通过标志：**校验失败仍为 400，认证失败仍为 401，响应正文与会话规则保持一致，成功／失败都有 `x-request-id`。

## 第 16 步：把你的项目控制器也改成原生参数校验

前面完成基础迁移，现在展示控制器代码真正变少的地方。这个步骤只移动 Body／Query 校验，保留权限策略、actor、ids 和 Service 调用。

先在运行后端的终端按 `Ctrl+C`。然后完整替换 `apps/server/src/modules/projects/projects.controller.ts`：

```ts
import { Body, Controller, Get, HttpCode, Patch, Post, Query, Req } from '@nestjs/common'
import type { FastifyRequest } from 'fastify'
import type {
  CreateProjectRequest,
  EntityPageQuery,
  SetAdministratorRequest,
  StatusRequest,
  UpdateProfileRequest,
} from 'shared/types'
import { AccessPolicy } from '@/common/decorators/access-policy.decorator'
import { actor, ids } from '@/modules/access/access-http'
import { entityListSchema, statusSchema } from '@/modules/access/access.validation'
import {
  createProjectSchema,
  profileSchema,
} from '@/modules/access/workspaces/workspaces.validation'
import { administratorSchema } from '@/modules/members/administrators/administrators.validation'
import { ProjectsService } from './projects.service'

@Controller()
export class ProjectsController {
  constructor(private readonly projectsService: ProjectsService) {}

  /** 校验项目资料并提交更新，路由身份和权限继续由守卫与 Service 检查。 */
  @Patch('companies/:companyId/projects/:projectId')
  @AccessPolicy({ scope: 'route', permission: '{scope}.profile.update' })
  updateProfile(
    @Req() req: FastifyRequest,
    @Body({ schema: profileSchema }) body: UpdateProfileRequest,
  ) {
    return this.projectsService.profile({
      actor: actor(req),
      companyId: ids(req).companyId!,
      projectId: ids(req).projectId!,
      body,
    })
  }

  /** 按权限和经转换的分页筛选参数读取项目列表。 */
  @Get('companies/:companyId/projects')
  @AccessPolicy({ scope: 'company', permission: 'company.projects.read' })
  projects(
    @Req() req: FastifyRequest,
    @Query({ schema: entityListSchema }) query: EntityPageQuery,
  ) {
    return this.projectsService.listProjects({
      actor: actor(req),
      companyId: ids(req).companyId!,
      query,
    })
  }

  /** 公司管理员创建项目，保留初始管理员和组织初始化业务流程。 */
  @Post('companies/:companyId/projects')
  @AccessPolicy({ scope: 'company', permission: 'company.projects.create', adminOnly: true })
  createProject(
    @Req() req: FastifyRequest,
    @Body({ schema: createProjectSchema }) body: CreateProjectRequest,
  ) {
    return this.projectsService.create({ actor: actor(req), companyId: ids(req).companyId!, body })
  }

  /** 根据既有项目权限读取项目资料与管理员信息。 */
  @Get('companies/:companyId/projects/:projectId')
  @AccessPolicy({ scope: 'project', permission: 'project.profile.read' })
  projectDetail(@Req() req: FastifyRequest) {
    const params = ids(req)
    return this.projectsService.get({
      actor: actor(req),
      companyId: params.companyId!,
      projectId: params.projectId!,
    })
  }

  /** 校验状态与预期版本后提交启停，业务权限与并发控制保持原样。 */
  @Patch('companies/:companyId/projects/:projectId/status')
  @AccessPolicy({ scope: 'company', permission: 'company.projects.status' })
  projectStatus(@Req() req: FastifyRequest, @Body({ schema: statusSchema }) body: StatusRequest) {
    const params = ids(req)
    return this.projectsService.status({
      actor: actor(req),
      companyId: params.companyId!,
      projectId: params.projectId!,
      body,
    })
  }

  /** 在公司管理员权限下设置项目管理员。 */
  @Post('companies/:companyId/projects/:projectId/administrator')
  @HttpCode(200)
  @AccessPolicy({ scope: 'company', permission: 'company.projects.administrator', adminOnly: true })
  projectAdministrator(
    @Req() req: FastifyRequest,
    @Body({ schema: administratorSchema }) body: SetAdministratorRequest,
  ) {
    const params = ids(req)
    return this.projectsService.setAdministrator({
      actor: actor(req),
      companyId: params.companyId!,
      projectId: params.projectId!,
      body,
    })
  }

  /** 返回公司管理员可以管理的项目管理员列表。 */
  @Get('companies/:companyId/projects/:projectId/administrators')
  @AccessPolicy({ scope: 'company', permission: 'company.projects.administrator', adminOnly: true })
  projectAdministrators(@Req() req: FastifyRequest) {
    const params = ids(req)
    return this.projectsService.projectAdministrators({
      actor: actor(req),
      companyId: params.companyId!,
      projectId: params.projectId!,
    })
  }
}
```

## 第 18 步：掌握以后新增接口的固定写法

下面是后续工作的规则说明，**本步不再要求修改现有业务文件**。认证和项目样板完成后，新增接口可以照它写。

1. 请求／响应类型放在 `packages/shared/src/types`，Schema 用 `satisfies z.ZodType<类型>` 对照。
2. Body 用 `@Body({ schema })`，整个 query 用 `@Query({ schema })`，整个路由参数用 `@Param({ schema })`。
3. 单个 ID 则是 `@Param('id', { schema: z.uuid() }) id: string`；不要把对象 Schema 用在字符串参数上。
4. 返回对象绑定 `@SerializeOptions({ schema: resultSchema })`，再绑定 `@ApiOkResponse({ standardSchema: resultSchema })` 或实际成功状态对应的响应装饰器。
5. 返回顶层数组时，原生拦截器按单项应用 Schema：`@SerializeOptions({ schema: itemSchema })`，Swagger 使用 `@ApiOkResponse({ standardSchema: itemSchema, isArray: true })`。
6. 分页对象是普通对象，例如 `{ items, total, page, pageSize }`，应该绑定整个分页对象 Schema；其中 items 才使用 `z.array(itemSchema)`。
7. `null`、`undefined`、顶层基本类型与 `StreamableFile` 会被当前原生拦截器直接放行；未绑定 Schema 的响应也放行。不要把它理解成自动验证所有返回值的开关。

其他尚未改造的控制器继续调用第 4 步的 `parse`，这不会阻塞 Nest 12 迁移。推广时每次选一个模块，用它原来的 Schema 替换参数装饰器并删除对应的 `parse` 调用；不改权限、事务和副作用。特别是账号重置密码的 `onTargetResolved` 安全事件回调需要保留；成员创建按作用域选 Schema 的条件分支，也不能直接换成某一个固定 Schema。

响应 Schema 必须描述接口实际公开字段。不能把数据库整行 Schema 直接挂到 API，也不能假定注册返回值跟项目详情相同。Zod 同一对象能服务“运行校验与文档生成”，TypeScript 类型负责开发期检查，两者仍是不同职责。
